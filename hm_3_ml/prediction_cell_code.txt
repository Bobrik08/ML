from sklearn.linear_model import LogisticRegression
import scipy.sparse as sp
from scipy.sparse import hstack, vstack

# Список «других» фичей как в base-ноутбуке (heroes + other_features)
other_features = ['region', 'avg_mmr', 'mmr_log']

matches_df_test = pd.read_csv("matches_df_test.csv")
if 'mmr_log' not in matches_df_test.columns:
    matches_df_test = matches_df_test.copy()
    matches_df_test['mmr_log'] = np.log(1 + matches_df_test['avg_mmr'])

# Собираем X_full в SPARSE формате, чтобы не конвертировать в плотный массив (~5.5 ГБ)
x = sp.load_npz('X_train.npz')
y = sp.load_npz('X_val.npz')
chat_train = dota_chat.set_index("match_id").reindex(y_train["match_id"])
chat_val = dota_chat.set_index("match_id").reindex(y_val["match_id"])
X_chat_train = hstack([vec_rad.transform(chat_train["radiant_clean"].fillna("")), vec_dir.transform(chat_train["dire_clean"].fillna(""))])
X_chat_val = hstack([vec_rad.transform(chat_val["radiant_clean"].fillna("")), vec_dir.transform(chat_val["dire_clean"].fillna(""))])
X_train_sparse = hstack([x, X_chat_train])
X_val_sparse = hstack([y, X_chat_val])
X_full = vstack([X_train_sparse, X_val_sparse])
y_full = np.concatenate([y_train.iloc[:, 1].values, y_val.iloc[:, 1].values])

model_full = LogisticRegression(C=48.8944801765785, solver="lbfgs", max_iter=3000)
model_full.fit(X_full, y_full)

# Энкодер героев нужно обучить на ТРЕЙНЕ (как в base), иначе колонки не совпадут
train_ids = y_train['match_id'].values
player_df_train = player_df[player_df['match_id'].isin(train_ids)].copy()
enc = HeroesEncoder()
enc.fit(player_df_train)

test_ids_ordered = matches_df_test['match_id'].values
player_df_test = player_df[player_df['match_id'].isin(test_ids_ordered)].copy()
order_map = pd.Series(range(len(test_ids_ordered)), index=matches_df_test['match_id'])
player_df_test = player_df_test.assign(_ord=player_df_test['match_id'].map(order_map)).sort_values('_ord').drop(columns=['_ord'])

X_heroes_test = enc.transform(player_df_test)
test_ids_in_encoder = player_df_test['match_id'].unique()

mid_to_row = {mid: i for i, mid in enumerate(test_ids_in_encoder)}
n_test = len(matches_df_test)
n_heroes = X_heroes_test.shape[1]
full_heroes = np.zeros((n_test, n_heroes))
for i, mid in enumerate(test_ids_ordered):
    if mid in mid_to_row:
        full_heroes[i] = X_heroes_test[mid_to_row[mid]].toarray().ravel()

X_heroes_test_full = csr_array(full_heroes)
gr_test = matches_df_test[other_features]
X_other_test = csr_array(gr_test)
chat_test = dota_chat.set_index("match_id").reindex(matches_df_test["match_id"])
X_rad_test = vec_rad.transform(chat_test["radiant_clean"].fillna(""))
X_dir_test = vec_dir.transform(chat_test["dire_clean"].fillna(""))
X_chat_test = hstack([X_rad_test, X_dir_test])
X_test_full = hstack([X_heroes_test_full, X_other_test, X_chat_test])
proba_test = model_full.predict_proba(X_test_full)[:, 1]
submission = pd.DataFrame({'ID': test_ids_ordered, 'Value': proba_test}).drop_duplicates(subset='ID', keep='first')
submission.to_csv('submission.csv', index=False)
